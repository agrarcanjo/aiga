cmake_minimum_required(VERSION 3.24)
project(AigaCaptureCompat VERSION 1.0.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR CMAKE_GENERATOR_PLATFORM MATCHES "ARM")
  message(FATAL_ERROR "Capture compatibility requires MSVC and native Windows x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
set(DETOURS ${CMAKE_CURRENT_SOURCE_DIR}/vendor/detours)
file(SHA256 ${CMAKE_CURRENT_SOURCE_DIR}/dll.cpp DLL_HASH)
file(SHA256 ${CMAKE_CURRENT_SOURCE_DIR}/protocol.h PROTOCOL_HASH)
string(SHA256 CAPTURE_BUILD_ID "${DLL_HASH}:${PROTOCOL_HASH}:e4bfd6b03e50de46b47abfbd1e46b384f0c5f833")
configure_file(build-id.h.in build-id.h @ONLY)
include_directories(${CMAKE_CURRENT_BINARY_DIR})
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY ${DETOURS} OUTPUT_VARIABLE DETOURS_REV OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT DETOURS_REV STREQUAL "e4bfd6b03e50de46b47abfbd1e46b384f0c5f833")
  message(FATAL_ERROR "Unexpected Detours revision; run build.ps1")
endif()
add_library(detours STATIC ${DETOURS}/src/detours.cpp ${DETOURS}/src/modules.cpp ${DETOURS}/src/disasm.cpp ${DETOURS}/src/image.cpp ${DETOURS}/src/creatwth.cpp)
target_include_directories(detours PUBLIC ${DETOURS}/src)
add_library(aiga-capture-compat SHARED dll.cpp)
target_link_libraries(aiga-capture-compat PRIVATE detours user32)
add_executable(aiga-capture-helper helper.cpp)
target_link_libraries(aiga-capture-helper PRIVATE user32)
add_executable(aiga-capture-fixture fixture.cpp)
target_link_libraries(aiga-capture-fixture PRIVATE user32 psapi)
foreach(target aiga-capture-compat aiga-capture-helper aiga-capture-fixture)
  target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /guard:cf)
  target_link_options(${target} PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf)
endforeach()
